<script>
function groupPubs(mode) {
  const container = document.getElementById("pub-list");
  const pubs = Array.from(container.querySelectorAll(".pub"));

  if (pubs.length === 0) return;

  // Remove existing group wrappers
  Array.from(container.querySelectorAll(".pub-group-wrapper")).forEach(w => w.remove());

  // Group pubs
  const groups = {};
  pubs.forEach(pub => {
    const key = mode === 'type' ? (pub.dataset.type || 'Unspecified') : (pub.dataset.topic || 'Unspecified');
    const year = parseInt(pub.dataset.year) || 0;
    if (!groups[key]) groups[key] = [];
    groups[key].push({ element: pub, year });
  });

  container.innerHTML = '';

  // Sort groups alphabetically
  Object.keys(groups).sort().forEach(key => {
    const wrapper = document.createElement("div");
    wrapper.className = "pub-group-wrapper";

    const header = document.createElement("div");
    header.className = "pub-group-header";
    header.textContent = key;
    wrapper.appendChild(header);

    // Sort publications by year descending
    groups[key].sort((a, b) => b.year - a.year).forEach(pub => {
      wrapper.appendChild(pub.element);
    });

    container.appendChild(wrapper);
  });

  // Re-init Bootstrap popovers if used
  if (typeof $ !== "undefined" && $.fn.popover) {
    $('[data-toggle="popover"]').popover();
  }
}

// Default grouping on page load
document.addEventListener("DOMContentLoaded", () => {
  groupPubs("type");
});
</script>
